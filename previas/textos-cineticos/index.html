Como usar

Três passos,
nenhum build.

  1. Copie estilo.css e script.js para o seu projeto.
  2. Marque o texto com data-tc="efeito".
  3. Pronto: o script inicia sozinho. Para controlar, use TextosCineticos.iniciar().
<!-- 1. estilos e script -->
<link rel="stylesheet" href="estilo.css">
<script src="script.js" defer></script>

<!-- 2. qualquer elemento de texto -->
<h2 data-tc="revelar">Seu título</h2>

<p data-tc="decodificar" data-duracao="1.5">
  Mensagem secreta
</p>

<span data-tc="digitar"
      data-frases="um|dois|três"></span>

<span data-tc="contador" data-valor="1284750"></span>

<!-- 3. opcional: controle manual -->
// <body data-tc-manual>
TextosCineticos.iniciar();
TextosCineticos.repetir(elemento);
